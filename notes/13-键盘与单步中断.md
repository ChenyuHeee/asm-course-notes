# 第13讲：键盘中断、单步中断与缓冲区溢出

> 本周核心：掌握键盘中断 int 9h 与扫描码处理，理解单步中断 int 1h 与 TF 陷阱标志，了解经典缓冲区溢出攻击原理。

---

## 一、键盘中断 int 9h

int 9h 是键盘硬件中断，按下或释放任意键时触发。

### 1.1 读取扫描码

键盘扫描码通过 **60h 端口** 读取：

```asm
in al, 60h          ; al = 扫描码
```

### 1.2 扫描码 bit 7：按下 vs 释放

- **bit 7 = 0**：按下（make code）
- **bit 7 = 1**：释放（break code）

```asm
test al, 80h        ; 检测 bit 7
jz down             ; bit 7 = 0 → 按下
; 否则是释放
```

> 释放码 = 按下码 + 80h。例如 Esc 的按下码是 `01h`，释放码是 `81h`。

### 1.3 扩展键盘前缀码

方向键、功能键等扩展键盘会先发一个前缀码 `0E0h` 或 `0E1h`，实际的键码跟在后面：

```asm
cmp al, 0E0h        ; 扩展键盘前缀
je extend
cmp al, 0E1h        ; 另一个扩展前缀
jne up_or_down
extend:
; 记录这是扩展键，等待下一个扫描码
```

### 1.4 完整示例：键盘扫描码显示器

每按下一个键，显示是按下还是释放，以及对应的扫描码：

```asm
data segment
old_9h dw 0, 0
stop   db 0
key    db 0
phead  dw 0
key_extend  db 'KeyExtend=', 0
key_up      db 'KeyUp=', 0
key_down    db 'KeyDown=', 0
key_code    db '00h ', 0
hex_tbl     db '0123456789ABCDEF'
cr          db 0Dh, 0Ah, 0
data ends

code segment
assume cs:code, ds:data
int_9h:
   push ax
   push bx
   push cx
   push ds
   mov ax, data
   mov ds, ax          ; ★ 必须设置 DS，中断时 DS 不一定是 data
   in al, 60h          ; 读取扫描码
   mov [key], al
   cmp al, 0E0h
   je  extend
   cmp al, 0E1h
   jne up_or_down
extend:
   mov [phead], offset key_extend
   call output
   jmp check_esc
up_or_down:
   test al, 80h        ; bit 7 = 0 → 按下，= 1 → 释放
   jz down
up:
   mov [phead], offset key_up
   call output
   mov bx, offset cr
   call display_str
   jmp check_esc
down:
   mov [phead], offset key_down
   call output
check_esc:   
   cmp [key], 81h      ; Esc 释放码 = 81h
   jne int_9h_iret
   mov [stop], 1       ; 通知主程序退出
int_9h_iret:
   mov al, 20h
   out 20h, al         ; 发送 EOI
   pop ds
   pop cx
   pop bx
   pop ax
   iret

output:                 ; 把 [key] 转成两位十六进制字符串
   push ax
   push bx
   push cx
   mov bx, offset hex_tbl
   mov cl, 4
   push ax
   shr al, cl          ; 取高 4 位
   xlat                ; AL = DS:[BX+AL]，查表转十六进制字符
   mov key_code[0], al
   pop ax
   and al, 0Fh         ; 取低 4 位
   xlat
   mov key_code[1], al
   mov bx, [phead]
   call display_str
   mov bx, offset key_code
   call display_str
   pop cx
   pop bx
   pop ax
   ret

display_str:            ; 用 int 10h / ah=0Eh 显示字符串
   push ax
   push bx
   push si
   mov si, bx
   mov bx, 0007h       ; bh = 页号, bl = 前景色
display_next_char:
   mov ah, 0Eh
   lodsb               ; AL = DS:[SI], SI++
   or al, al
   jz display_str_done
   int 10h
   jmp display_next_char
display_str_done:
   pop si
   pop bx
   pop ax
   ret

main:
   mov ax, data
   mov ds, ax
   cld                  ; DF=0，字符串操作向前
   xor ax, ax
   mov es, ax
   mov bx, 9*4
   ; 保存旧向量
   push word ptr es:[bx]
   pop old_9h[0]
   push word ptr es:[bx+2]
   pop old_9h[2]
   cli
   mov word ptr es:[bx], offset int_9h
   mov es:[bx+2], cs
   sti
wait_again:
   cmp [stop], 1
   jne wait_again      ; 循环等待，直到 Esc 释放
   ; 恢复旧向量
   push old_9h[0]
   pop es:[bx]
   push old_9h[2]
   pop es:[bx+2]
   mov ah, 4Ch
   int 21h
code ends
end main
```

> 关键点：
> - 中断处理函数第一件事 `mov ax, data; mov ds, ax`，因为中断发生时 DS 不一定是自己的数据段
> - `xlat` 指令 = `mov al, ds:[bx+al]`，用于查表转换（二进制 → 十六进制字符）
> - `int 10h / ah=0Eh` 是 BIOS 的「电传打字机输出」，逐字符显示，自动处理换行

## 二、单步中断 int 1h 与 TF 陷阱标志

### 2.1 FL 标志寄存器

FL（FLAGS）寄存器包含以下常用标志位：

| 位 | 标志 | 全称 | 含义 |
|:---|:-----|:-----|:-----|
| 0 | CF | Carry | 进位/借位 |
| 6 | ZF | Zero | 结果为零 |
| 7 | SF | Sign | 符号（=最高位） |
| 11 | OF | Overflow | 有符号溢出 |
| 8 | **TF** | **Trap** | **单步模式** |

TF（Trap Flag）是 FL 的第 8 位，值 = `100h`。**当 TF=1 时，CPU 每执行一条指令后自动触发 int 1h**。

### 2.2 设置和清除 TF

```asm
; --- 设置 TF=1（进入单步模式）---
pushf                 ; FL 入栈
pop ax                ; ax = FL
or ax, 100h           ; 第 8 位置 1
push ax
popf                  ; FL = ax（注意：popf 本身不会触发 int 1h）
                      ; ★ popf 的下一条指令执行后才会触发 int 1h

; --- 清除 TF=0（退出单步模式）---
pushf
pop ax
and ax, 0FEFFh        ; 第 8 位清零（0FEFFh = not 100h）
push ax
popf
```

### 2.3 示例1：空处理 int 1h

最简单的单步中断演示——拦截 int 1h 但不做额外操作：

```asm
code segment
assume cs:code
old_1h dw 0, 0

int_1h:
   nop                ; 什么都不做
   iret               ; CPU 进入中断时会自动清除 IF 和 TF

main:
   xor ax, ax
   mov es, ax
   mov bx, 4          ; int 1h → 向量在 0:4
   mov ax, es:[bx]
   mov dx, es:[bx+2]
   mov cs:old_1h[0], ax
   mov cs:old_1h[2], dx
   mov word ptr es:[bx], offset int_1h
   mov es:[bx+2], cs
   ; 进入单步模式
   pushf
   pop ax
   or ax, 100h
   push ax
   popf               ; TF=1
   ; 以下每条指令执行后都会触发 int 1h
   nop                ; ← int 1h 在此触发
   mov ax, 0          ; ← int 1h
   mov bx, 1          ; ← int 1h
again:
   add ax, bx         ; ← int 1h
   cmp ax, 100        ; ← int 1h
   jb again           ; ← int 1h
   popf               ; TF=0（此处还会触发一次 int 1h）
   nop
   ; 退出单步模式
   mov ax, cs:old_1h[0]
   mov dx, cs:old_1h[2]
   mov es:[bx], ax
   mov es:[bx+2], dx
   mov ah, 4Ch
   int 21h
code ends
end main
```

> 注意：CPU 进入中断时会自动清零 TF（和 IF），所以 `int_1h` 内部的指令不会触发嵌套的单步中断。`iret` 恢复 FL 后 TF 重新 = 1，下一条指令执行后又触发 int 1h。

### 2.4 示例2：代码自修改（反调试技术雏形）

自定义 int 1h 处理函数中，**访问堆栈中的返回地址并修改原指令的字节码**：

```asm
code segment
assume cs:code
old_1h dw 0, 0
prev   dw offset first, seg first

int_1h:
   push bp
   mov bp, sp
   push es
   push di
   push ds
   push si
   les di, dword ptr [bp+2]  ; ES:DI = 触发 int 1h 的那条指令的地址
   dec byte ptr es:[di]      ; ★ 把触发位置的指令字节减 1
   lds si, dword ptr [prev]  ; DS:SI = 上一条被修改的指令地址
   inc byte ptr ds:[si]      ; ★ 把上一条指令的字节恢复（加回来）
   mov prev[0], di           ; 记录本次修改的位置
   mov prev[2], es
   pop si
   pop ds
   pop di
   pop es
   pop bp
   iret

main:
   xor ax, ax
   mov es, ax
   mov bx, 4
   mov ax, es:[bx]
   mov dx, es:[bx+2]
   mov cs:old_1h[0], ax
   mov cs:old_1h[2], dx
   mov word ptr es:[bx], offset int_1h
   mov es:[bx+2], cs
   pushf
   pop ax
   or ax, 100h
   push ax
   popf                  ; TF=1
first:   
   nop                   ; ← int 1h 触发，int_1h 修改此指令字节
   ; --- 以下指令的字节码需要在编译后用 QV 修改为原值+1 ---
   mov ax, 0             ; ← int 1h 触发，int_1h 把它恢复成正确的字节
   mov bx, 1
again:
   add ax, bx
   cmp ax, 100
   jb again
   popf                  ; TF=0
   nop
   ; --- 以上指令修改结束 ---
   mov ax, cs:old_1h[0]
   mov dx, cs:old_1h[2]
   mov es:[bx], ax
   mov es:[bx+2], dx
   mov ah, 4Ch
   int 21h
code ends
end main
```

> 原理：将关键指令的字节码预先 +1（编译后用 QV 修改），运行时 int 1h 处理函数自动把 +1 的字节减回去，恢复为正确指令。如果有人在单步调试，int 1h 被替换为空处理，指令就不会被修复，程序崩溃——这是一种**反单步调试**技术。

## 三、缓冲区溢出攻击演示

### 3.1 漏洞函数

```cpp
int len(char *p)
{
   char t[4];           // 只分配了 4 字节
   strcpy(t, p);        // 不做边界检查！
   return strlen(t);
}
```

### 3.2 栈帧布局

```
ebp-4  → t[0..3]       4字节局部数组
ebp+0  → old ebp       (被覆盖)
ebp+4  → 返回地址      (★ 被劫持！覆盖为 jmp esp 的地址)
ebp+8  → 参数 p        (指向 shellcode)
```

### 3.3 攻击 payload 结构

```
[t[0..3]]  [old ebp]   [返回地址]      [参数 p 指向的数据]
4×0x90  + 4×0x90    + 0x7FFA4512   + shellcode...
```

- 前 8 字节：NOP（0x90）填充
- 返回地址：覆盖为 `0x7FFA4512`，该处有一条 `jmp esp` 指令
- shellcode：`mov ebp, esp; xor eax, eax; ...` → 调用 `MessageBox` 弹窗

### 3.4 攻击流程

1. `strcpy(t, p)` 溢出 t[4]，覆盖 old ebp 和返回地址
2. `len()` 执行 `ret`，返回到 `0x7FFA4512`
3. `0x7FFA4512` 处是 `jmp esp`，跳到 esp 当前位置
4. esp 当前指向参数 p 中的 shellcode（在返回地址之上）
5. shellcode 执行，弹出 "Buffer Overflow!" 消息框

> 这是一个经典的栈溢出利用模式：**NOP sled + return address + shellcode**。`jmp esp` gadget 在现代漏洞利用中被称为 ROP 的前身。

## 四、本周易错点

1. int 9h 处理函数中忘记 `mov ax, data; mov ds, ax`，导致访问错误的内存
2. 扫描码 bit 7 判断按下/释放时用错指令（应 `test al, 80h; jz down`）
3. 硬件中断不止 int 8h，int 9h 也要发 EOI
4. TF 设置的 `or ax, 100h` 写成 `or ax, 100`（十进制 100 ≠ 第8位）
5. 单步模式下 `popf` 后**不会立刻触发** int 1h，而是下一条指令执行后触发

## 五、速查表

| 操作 | 写法 |
|:-----|:-----|
| 读扫描码 | `in al, 60h` |
| 判断按下/释放 | `test al, 80h; jz down` |
| 设置 TF=1 | `pushf; pop ax; or ax, 100h; push ax; popf` |
| 清除 TF=0 | `pushf; pop ax; and ax, 0FEFFh; push ax; popf` |
| 查表转换 | `xlat`（bx 指向表，al 为索引） |
| BIOS 电传输出 | `mov ah, 0Eh; int 10h` |
| 取触发 int 1h 的地址 | `les di, [bp+2]`（中断栈帧中） |
